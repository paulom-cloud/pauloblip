<script>
  const stages = [
    { id: 'stage-0', tab: 0, label: 'DESCOBERTA\nFunil WhatsApp' },
    { id: 'stage-1', tab: 1, label: 'QUALIFICAÇÃO\nAgente IA + Flows' },
    { id: 'stage-2', tab: 2, label: 'CONVERSÃO\nCheckout PIX' },
    { id: 'stage-3', tab: 3, label: 'ONBOARDING\nEngajamento' },
    { id: 'stage-4', tab: 4, label: 'RETENÇÃO\nCobrança + NPS' },
    { id: 'stage-5', tab: 5, label: 'EXPANSÃO\nUpsell + Indicação' },
  ];
  let current = 0;

  function goToStage(idx) {
    document.querySelectorAll('.stage-screen').forEach(s => s.classList.remove('active'));
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    current = idx;
    document.getElementById(stages[idx].id).classList.add('active');
    document.querySelectorAll('.tab')[idx].classList.add('active');
    document.getElementById('stageLabel').textContent = stages[idx].label;
    document.getElementById('navProgress').textContent = (idx + 1) + ' / ' + stages.length;
    document.getElementById('prevBtn').disabled = idx === 0;
    document.getElementById('nextBtn').disabled = idx === stages.length - 1;
    document.querySelectorAll('.tab')[idx].scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
    setTimeout(() => {
      const bg = document.getElementById(stages[idx].id).querySelector('.wa-messages-bg');
      if (bg) bg.scrollTop = bg.scrollHeight;
    }, 100);
  }

  function navigate(dir) {
    const next = current + dir;
    if (next >= 0 && next < stages.length) goToStage(next);
  }

  goToStage(0);
</script>
